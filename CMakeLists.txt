cmake_minimum_required(VERSION 3.16)
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "Release C flags")
project(WBSM4GFNI LANGUAGES C)
include(CTest)
foreach(pair "XL;XL-WBSM4" "MASKED;Masked-WBSM4" "BAIWU;Bai-Wu-WBSM4" "WSISE;WSISE" "WBDL;WBDL" "CHEN;Chen-SE-WBSM4" "GONG;Gong-SE-WBSM4")
  list(GET pair 0 key)
  list(GET pair 1 folder)
  option(BUILD_${key} "Build ${folder}" OFF)
  if(BUILD_${key})
    add_subdirectory(${folder})
  endif()
endforeach()
