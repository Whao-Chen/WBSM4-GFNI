cmake_minimum_required(VERSION 3.16)
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "Release C flags")
project(MaskedWBSM4 LANGUAGES C)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
find_package(OpenSSL REQUIRED)
if(NOT TARGET WBMatrix)
  add_subdirectory("../BackEnd" "BackEnd")
endif()
include("../BackEnd/options.cmake")

function(release_programs implementation width)
  foreach(direction encrypt decrypt)
    if(direction STREQUAL "decrypt")
      set(dec 1)
    else()
      set(dec 0)
    endif()
    add_executable(test_${implementation}_${direction}
      test/correctness.c test/adapter_common.c test/release_adapter.c)
    add_executable(bench_${implementation}_${direction}
      test/throughput_common.c test/adapter_common.c test/release_adapter.c)
    foreach(prefix test bench)
      set(program ${prefix}_${implementation}_${direction})
      target_include_directories(${program} PRIVATE test)
      target_link_libraries(${program} PRIVATE ${implementation} OpenSSL::Crypto)
      target_compile_definitions(${program} PRIVATE BENCH_DECRYPT=${dec}
        BENCH_ALGORITHM="${PROJECT_NAME}" BENCH_VARIANT="${implementation}"
        BENCH_PATH_KIND="complete-call" BENCH_WIDTH=${width} BENCH_SEED=0x26082401u
        RELEASE_REQUIRE_WIDE=1)
      wbs_compile_options(${program} wide)
    endforeach()
    add_test(NAME ${implementation}_${direction} COMMAND test_${implementation}_${direction})
    set_tests_properties(${implementation}_${direction} PROPERTIES SKIP_RETURN_CODE 77)
  endforeach()
  add_executable(bench_${implementation}_construction
    test/bench_construction.c test/release_adapter.c)
  target_include_directories(bench_${implementation}_construction PRIVATE test)
  target_link_libraries(bench_${implementation}_construction PRIVATE ${implementation})
  target_compile_definitions(bench_${implementation}_construction PRIVATE RELEASE_REQUIRE_WIDE=1)
  wbs_compile_options(bench_${implementation}_construction wide)
endfunction()

foreach(variant baseline backend optimized construction_optimized)
  if(variant STREQUAL "baseline")
    set(source src/simplified/scalar.c)
    set(back WBMatrix_xl)
  elseif(variant STREQUAL "backend")
    set(source src/simplified/scalar.c)
    set(back WBMatrixGFNI_xl)
  elseif(variant STREQUAL "construction_optimized")
    set(source src/simplified/construction_parallel.c)
    set(back WBMatrixGFNI_xl)
  else()
    set(source src/simplified/parallel.c)
    set(back WBMatrixGFNI_xl)
  endif()
  add_library(masked_${variant} STATIC ${source} src/common/sm4.c)
  target_include_directories(masked_${variant} PUBLIC include)
  target_link_libraries(masked_${variant} PUBLIC ${back})
  if(variant STREQUAL "backend")
    target_compile_definitions(masked_${variant} PRIVATE WBSM4_GFNI_BACKEND=1)
  endif()
  if(variant STREQUAL "optimized")
    target_compile_definitions(masked_${variant} INTERFACE RELEASE_HAVE_MANY=1)
    set(width 16)
  else()
    set(width 1)
  endif()
  wbs_compile_options(masked_${variant} wide)
  release_programs(masked_${variant} ${width})
endforeach()
foreach(variant baseline backend optimized)
  if(variant STREQUAL "baseline")
    set(source src/enhanced/scalar.c)
    set(back WBMatrix_xl)
  elseif(variant STREQUAL "backend")
    set(source src/enhanced/scalar.c)
    set(back WBMatrixGFNI)
  else()
    set(source src/enhanced/parallel.c)
    set(back WBMatrixGFNI)
  endif()
  add_library(masked_enhanced_${variant} STATIC ${source} src/common/sm4.c)
  target_include_directories(masked_enhanced_${variant} PUBLIC include)
  target_link_libraries(masked_enhanced_${variant} PUBLIC ${back})
  target_compile_definitions(masked_enhanced_${variant} INTERFACE RELEASE_ENHANCED=1)
  if(variant STREQUAL "optimized")
    target_compile_definitions(masked_enhanced_${variant} PRIVATE
      MASKED_GFNI_TABLE2=1 MASKED_GFNI_VERIFY_TABLE2=0
      MASKED_PRECOMPOSE_MASK_LOOKUP_U8=1 MASKED_PRECOMPOSE_MASK_BRANCH=0
      MASKED_GFNI_VERIFY_MASK_BRANCH=0 MASKED_FUSE_U8_MASK_AFFINE=0 MASKED_PIPELINE_32WAY=0)
    target_compile_definitions(masked_enhanced_${variant} INTERFACE RELEASE_HAVE_MANY=1)
    set(width 16)
  else()
    set(width 1)
  endif()
  wbs_compile_options(masked_enhanced_${variant} wide)
  release_programs(masked_enhanced_${variant} ${width})
endforeach()
