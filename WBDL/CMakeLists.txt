cmake_minimum_required(VERSION 3.16)
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "Release C flags")
project(WBDL LANGUAGES C)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
find_package(OpenSSL REQUIRED)
if(NOT TARGET WBMatrix)
  add_subdirectory("../BackEnd" "BackEnd")
endif()
include("../BackEnd/options.cmake")

function(release_programs implementation width)
  foreach(direction encrypt decrypt)
    if(direction STREQUAL "decrypt")
      set(dec 1)
    else()
      set(dec 0)
    endif()
    add_executable(test_${implementation}_${direction}
      test/correctness.c test/adapter_common.c test/release_adapter.c)
    add_executable(bench_${implementation}_${direction}
      test/throughput_common.c test/adapter_common.c test/release_adapter.c)
    foreach(prefix test bench)
      set(program ${prefix}_${implementation}_${direction})
      target_include_directories(${program} PRIVATE test)
      target_link_libraries(${program} PRIVATE ${implementation} OpenSSL::Crypto)
      target_compile_definitions(${program} PRIVATE BENCH_DECRYPT=${dec}
        BENCH_ALGORITHM="${PROJECT_NAME}" BENCH_VARIANT="${implementation}"
        BENCH_PATH_KIND="complete-call" BENCH_WIDTH=${width} BENCH_SEED=0x26082401u
        RELEASE_REQUIRE_WIDE=1)
      wbs_compile_options(${program} wide)
    endforeach()
    add_test(NAME ${implementation}_${direction} COMMAND test_${implementation}_${direction})
    set_tests_properties(${implementation}_${direction} PROPERTIES SKIP_RETURN_CODE 77)
  endforeach()
  add_executable(bench_${implementation}_construction
    test/bench_construction.c test/release_adapter.c)
  target_include_directories(bench_${implementation}_construction PRIVATE test)
  target_link_libraries(bench_${implementation}_construction PRIVATE ${implementation})
  target_compile_definitions(bench_${implementation}_construction PRIVATE RELEASE_REQUIRE_WIDE=1)
  wbs_compile_options(bench_${implementation}_construction wide)
endfunction()

foreach(variant baseline backend optimized)
  if(variant STREQUAL "optimized")
    set(files src/parallel.c src/wbdl_gfni.c)
    set(back WBMatrixGFNI_legacy)
  else()
    set(files src/scalar.c)
    if(variant STREQUAL "baseline")
      set(back WBMatrix_wbdl)
    else()
      set(back WBMatrixGFNI_legacy)
    endif()
  endif()
  add_library(wbdl_${variant} STATIC ${files} src/sm4.c)
  target_include_directories(wbdl_${variant} PUBLIC include)
  if(variant STREQUAL "optimized")
    target_include_directories(wbdl_${variant} BEFORE PUBLIC include/parallel)
  elseif(variant STREQUAL "backend")
    target_compile_definitions(wbdl_${variant} PRIVATE WBSM4_GFNI_BACKEND=1)
  endif()
  target_compile_definitions(wbdl_${variant} INTERFACE RELEASE_DYNAMIC=1)
  target_link_libraries(wbdl_${variant} PUBLIC ${back})
  wbs_compile_options(wbdl_${variant} wide)
  release_programs(wbdl_${variant} 1)
endforeach()
